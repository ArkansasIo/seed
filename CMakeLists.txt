cmake_minimum_required(VERSION 3.25)
project(SeedEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SEED_WITH_MYSQL "Build with MySQL C client library" OFF)
option(SEED_BUILD_TESTS "Build unit tests" ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(seed_core src/core/config.cpp src/core/crypto.cpp src/core/database.cpp src/core/engine.cpp src/core/protocol.cpp src/core/socket.cpp src/core/ui.cpp)
target_include_directories(seed_core PUBLIC include)
target_link_libraries(seed_core PUBLIC Threads::Threads OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(seed_core PUBLIC ws2_32)
endif()
if(SEED_WITH_MYSQL)
  find_path(MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql mariadb)
  find_library(MYSQL_CLIENT_LIBRARY NAMES mysqlclient mariadb)
  if(NOT MYSQL_INCLUDE_DIR OR NOT MYSQL_CLIENT_LIBRARY)
    message(FATAL_ERROR "MySQL client headers/library not found")
  endif()
  target_include_directories(seed_core PRIVATE ${MYSQL_INCLUDE_DIR})
  target_link_libraries(seed_core PRIVATE ${MYSQL_CLIENT_LIBRARY})
  target_compile_definitions(seed_core PUBLIC SEED_WITH_MYSQL=1)
endif()
if(MSVC)
  target_compile_options(seed_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(seed_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(seed-server src/server/main.cpp)
target_link_libraries(seed-server PRIVATE seed_core)
add_executable(seed-client src/client/main.cpp)
target_link_libraries(seed-client PRIVATE seed_core)
add_executable(seed-admin src/admin/main.cpp)
target_link_libraries(seed-admin PRIVATE seed_core)
if(SEED_BUILD_TESTS)
  enable_testing()
  add_executable(seed-tests tests/test_main.cpp)
  target_link_libraries(seed-tests PRIVATE seed_core)
  add_test(NAME seed-tests COMMAND seed-tests)
endif()
install(TARGETS seed-server seed-client seed-admin RUNTIME DESTINATION bin)
